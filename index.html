1. Tiny C Language Overview

                  

                    Tiny C is an expressive subset of the C programming language designed for compiler design courses. It incorporates fundamental imperative programming constructs including data types (int, char, void), multi-dimensional control structures (if/else, while, for), 1D arrays, and recursive function definitions.

3. BNF / EBNF Formal Grammar Specification

                  
<program>             ::= <declaration>* EOF
<declaration>         ::= <type> IDENTIFIER ( <function_decl> | <variable_decl> )
<type>                ::= "int" | "char" | "void"
<function_decl>       ::= "(" <param_list>? ")" <block>
<param_list>          ::= <param> ( "," <param> )*
<param>               ::= <type> IDENTIFIER ( "[" "]" )?
<variable_decl>       ::= ( "[" INT_LITERAL "]" )? ( "=" <expression> )? ";"

<statement>           ::= <block>
                        | "if" "(" <expression> ")" <statement> ( "else" <statement> )?
                        | "while" "(" <expression> ")" <statement>
                        | "for" "(" <for_init> ";" <expression>? ";" <expression>? ")" <statement>
                        | "return" <expression>? ";"
                        | "break" ";"
                        | "continue" ";"
                        | <type> IDENTIFIER <variable_decl>
                        | <expression> ";"

<expression>          ::= <assignment>
<assignment>          ::= <logical_or> ( ( "=" | "+=" | "-=" | "*=" | "/=" ) <assignment> )?
<logical_or>          ::= <logical_and> ( "||" <logical_and> )*
<logical_and>         ::= <equality> ( "&&" <equality> )*
<equality>            ::= <relational> ( ( "==" | "!=" ) <relational> )*
<relational>          ::= <additive> ( ( "<" | "<=" | ">" | ">=" ) <additive> )*
<additive>            ::= <multiplicative> ( ( "+" | "-" ) <multiplicative> )*
<multiplicative>      ::= <unary> ( ( "*" | "/" | "%" ) <unary> )*
<unary>               ::= ( "!" | "-" | "+" | "++" | "--" ) <unary> | <postfix>
<postfix>             ::= <primary> ( "[" <expression> "]" | "(" <arg_list>? ")" | "++" | "--" )*
<primary>             ::= INT_LITERAL | CHAR_LITERAL | STRING_LITERAL | IDENTIFIER | "(" <expression> ")"
